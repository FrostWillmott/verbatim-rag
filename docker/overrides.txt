# Resolver overrides for the container dependency lock.
#
# docker/constraints.txt is the known-working environment for the Docker image.
# It is generated, not hand-edited. This file is the hand-maintained input: it
# lists resolutions that are known to break the container and must be excluded.
#
# The library's own supported ranges stay in pyproject.toml and are deliberately
# not narrowed here — this file only constrains what goes into the image.
#
# Regenerate after changing dependencies in pyproject.toml, or when refreshing
# the pinned set:
#
#   uv pip compile pyproject.toml --universal --python-version 3.11 \
#       --override docker/overrides.txt --no-emit-package verbatim-core \
#       -o docker/constraints.txt
#
# python-version must match the base image in Dockerfile (python:3.11-slim).
# --universal resolves with environment markers instead of for the host, so the
# same lock builds on both arm64 and x86_64 machines; without it the image only
# builds on the architecture it was generated on.
#
# --no-emit-package verbatim-core is required. [tool.uv.sources] in pyproject.toml
# points that name at the local packages/core so development uses the code being
# edited rather than the published release; without this flag the compile writes
# `-e packages/core` into constraints.txt, and pip rejects editable entries inside
# a constraints file — the image then fails to build. constraints.txt therefore no
# longer pins verbatim-core; the Dockerfile installs ./packages/core directly, so
# the pin was informational.
#
# After regenerating, verify rather than trust the resolution:
#
#   docker compose up --build
#   curl -fsS http://localhost/api/status

# milvus-lite 3.x rejects collections that have no vector field:
#
#   MilvusException: (code=6, message=schema has no vector field
#   (FLOAT_VECTOR or SPARSE_FLOAT_VECTOR))
#
# verbatim_rag/vector_stores/milvus_local.py creates exactly that for the
# document-metadata collection ("no vectors, just metadata"), so RAG init
# fails and /api/status returns 500. Verified: the same create_collection
# succeeds on milvus-lite 2.5.1 and fails on 3.1.0, on both pymilvus 2.6.17
# and 3.0.0 — so this is a milvus-lite regression, not a pymilvus API change.
milvus-lite<3

# milvus-lite 2.x imports pkg_resources, which setuptools removed in 81:
#
#   File "/usr/local/lib/python3.11/site-packages/milvus_lite/__init__.py", line 15
#     from pkg_resources import DistributionNotFound, get_distribution
#   ModuleNotFoundError: No module named 'pkg_resources'
#
# pymilvus reports this as "milvus-lite is required for local database
# connections", RAG init fails and /api/status returns 500. It stays hidden
# with torch 2.6.0, which requires setuptools only under python_version >=
# "3.12" and so left the base image's older setuptools in place on 3.11;
# torch 2.13.0 requires setuptools>=77.0.3 unconditionally and pulls in 83.x.
# 80.x satisfies torch and still ships pkg_resources.
#
# What retires this pin, since a calendar date on it would only say when someone
# promised to look: milvus-lite dropping the pkg_resources import. Both pins go
# together — `milvus-lite<3` above is the reason this one exists — so the check
# is a single command against the unpinned pair:
#
#   pip install -q "milvus-lite>=3" "setuptools>=81" &&
#     python -c "import milvus_lite; print(milvus_lite.__version__)"
#
# If that imports cleanly, re-test the metadata collection from
# milvus_local.py before removing either line: the schema regression documented
# above is the other half, and it is not fixed by the import working.
setuptools<81
